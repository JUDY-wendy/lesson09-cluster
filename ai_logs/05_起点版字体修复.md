```python
from matplotlib import font_manager

available_fonts = {f.name for f in font_manager.fontManager.ttflist}
chosen_font = next(
    (f for f in ("Microsoft YaHei", "SimHei") if f in available_fonts), None
)
if chosen_font is None:
    raise SystemExit("未找到可用中文字体（Microsoft YaHei / SimHei），停止运行")
matplotlib.rcParams["font.sans-serif"] = [chosen_font, "DejaVu Sans"]

print(f"本次绘图使用中文字体：{chosen_font}")
```
